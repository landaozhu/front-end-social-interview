### https://leetcode.cn/problems/longest-consecutive-sequence/description/?envType=study-plan-v2&envId=top-100-liked
function maxArrLength(nums: number[]) {
let maxLen = -Infinity
// hash { }
const hash = new Set()
for (let i = 0; i <nums.length; i++) {
hash.add(nums[i])
}
// 从前往后如果存在 hash [nums[i]] - 1 则无需判断，否则 while ++ maxlength
for (let i = 0; i < nums.length; i++) {
if (hash.has(nums[i] - 1)) {
continue
}
let tempLength = 1
while (hash.has(nums[i] + 1)) {
tempLength++
}
maxLen = Math.max(maxLen, tempLength)
}
return maxLen
}console.log(maxArrLength([100,4,200,1,3,2]))
